+"""Tests for the PILOT-045 command layer: ``campaign-check`` and ``brief``.
+
+Every case builds its own throw-away tower root under the runner's ``TMP``
+sandbox and drives the real CLI through ``subprocess``, so exit codes and
+stdout are asserted exactly as a caller observes them::
+
+    python -B tools/run_tests.py

+The YAML-subset cases call the parser directly; the rest go through
+``towerctl campaign-check`` / ``towerctl brief`` so the wiring is covered too.

+def campaign_text(task_forces: tuple[str, ...] = (TASK,), goal: str = "G2",

+def run_cli(root: Path, *args: str) -> subprocess.CompletedProcess:
+    return subprocess.run(
+        [sys.executable, "-B", str(TCLI), "--root", str(root), *args],
+        capture_output=True, text=True, encoding="utf-8")

+def write_json(path: Path, data) -> None:
+    path.parent.mkdir(parents=True, exist_ok=True)
+    path.write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")

+class CommandLayerCase(unittest.TestCase):
+    """A minimal tower root: strategy, one campaign, board, state, one record."""
+
+    def setUp(self) -> None:
+        self._tmp = tempfile.TemporaryDirectory(
+            prefix="towerctl-command-", ignore_cleanup_errors=True)
+        self.tower = Path(self._tmp.name) / "tower"
+        self.write_doc(STRATEGY_REL, strategy_text())
+        self.write_doc(CAMPAIGN_REL, campaign_text())
+        self.write_board()
+        write_json(self.tower / "primary" / "CURRENT_STATE.json", {
+            "schema": "tower/current-state@1", "owner": {"name": "mac"},
+            "next_action": "跑 045", "blockers": []})
+        self.write_record(TASK, "dispatched")
+
+    def tearDown(self) -> None:
+        self._tmp.cleanup()
+
+    def write_doc(self, relpath: str, text: str) -> Path:
+        path = self.tower / relpath
+        path.parent.mkdir(parents=True, exist_ok=True)
+        path.write_text(text, encoding="utf-8")
+        return path
+
+    def write_record(self, task_id: str, status: str,
+                     verdict: str | None = None) -> None:
+        write_json(self.tower / "tasks" / task_id / "record.json", {
+            "schema": "tower/task@1", "id": task_id, "title": "夹具卡",
+            "status": status, "next": "等待处理",
+            "review": {"verdict": verdict} if verdict else None})
+
+    def write_board(self, *task_ids: str) -> None:
+        write_json(self.tower / "primary" / "TASK_BOARD.json", {
+            "schema": "tower/task-board@1",
+            "tasks": [{"id": task_id, "status": "closed"} for task_id in task_ids]})
+
+    def check(self, *args: str) -> subprocess.CompletedProcess:
+        return run_cli(self.tower, "campaign-check", *args)
+
+    def payload(self, *args: str) -> tuple[subprocess.CompletedProcess, dict]:
+        cp = self.check(*args)
+        return cp, json.loads(cp.stdout) if cp.stdout.strip() else {}
+
+    def codes(self, payload: dict, key: str = "errors") -> list[str]:
+        return [finding["code"] for finding in payload.get(key) or []]
+
+    def snapshot(self) -> dict[str, str]:
+        return {path.relative_to(self.tower).as_posix():
+                hashlib.sha256(path.read_bytes()).hexdigest()
+                for path in sorted(self.tower.rglob("*")) if path.is_file()}

+def front_matter_block(text: str) -> list[str]:
+    """The lines between the leading and the closing ``---`` fence."""
+    lines = text.splitlines()
+    if not lines or lines[0].strip() != "---":
+        raise CommandDocError("front-matter must start on the first line with ---")
+    for index in range(1, len(lines)):
+        if lines[index].strip() == "---":
+            return lines[1:index]
+    raise CommandDocError("front-matter has no closing ---")

+def yaml_scalar(raw: str) -> str:
+    """A plain scalar or a double-quoted one; no flow syntax, no escapes."""
+    text = raw.strip()
+    if not text:
+        raise CommandDocError("empty value where a scalar is required")
+    if text.startswith('"'):
+        if len(text) < 2 or not text.endswith('"'):
+            raise CommandDocError(f"unterminated quoted string: {text!r}")
+        if '"' in text[1:-1]:
+            raise CommandDocError(f"escaped quotes are unsupported: {text!r}")
+        return text[1:-1]
+    if text[0] in "[{&*!|>%@`" or text.endswith('"'):
+        raise CommandDocError(f"unsupported scalar syntax: {text!r}")
+    return text
+

+def yaml_key_value(text: str, indent: int) -> tuple[str, str]:
+    """Split ``key: value`` at ``indent`` spaces; reject anything else."""
+    if text[:indent].strip():
+        raise CommandDocError(f"unexpected indentation: {text!r}")
+    body = text[indent:]
+    if ":" not in body:
+        raise CommandDocError(f"expected 'key: value': {body!r}")
+    key, _, value = body.partition(":")
+    key = key.strip()
+    if not COMMAND_KEY_RE.match(key):
+        raise CommandDocError(f"unsupported key: {key!r}")
+    return key, value.strip()
+

+def yaml_mapping_item(rest: str) -> bool:
+    """True when a list item body is ``key: value`` rather than a scalar."""
+    head = rest.split(":", 1)[0].strip()
+    return ":" in rest and bool(COMMAND_KEY_RE.match(head))

+def yaml_mapping_entry(lines: list[str], index: int,
+                       head: str) -> tuple[dict, int]:
+    """Parse ``- key: value`` plus its 4-space continuation keys."""
+    key, value = yaml_key_value(head, 0)
+    item = {key: yaml_scalar(value)}
+    index += 1
+    while index < len(lines):
+        line = lines[index]
+        if not line.strip() or line.lstrip().startswith("#"):
+            index += 1
+            continue
+        if line.startswith("     "):
+            raise CommandDocError(f"unsupported nested structure: {line.strip()!r}")
+        if not line.startswith("    "):
+            break
+        key, value = yaml_key_value(line, 4)
+        if key in item:
+            raise CommandDocError(f"duplicate key in mapping item: {key}")
+        item[key] = yaml_scalar(value)
+        index += 1
+    return item, index

+def yaml_list_block(lines: list[str], start: int) -> tuple[list, int]:
+    """Parse the 2-space list that follows a bare ``key:``."""
+    entries: list = []
+    index = start
+    mode: bool | None = None
+    while index < len(lines):
+        line = lines[index]
+        if not line.strip() or line.lstrip().startswith("#"):
+            index += 1
+            continue
+        if not line.startswith("  - "):
+            break
+        body = line[4:]
+        mapping = yaml_mapping_item(body)
+        if mode is None:
+            mode = mapping
+        elif mode is not mapping:
+            raise CommandDocError("mixed scalar and mapping list items are unsupported")
+        if mapping:
+            entry, index = yaml_mapping_entry(lines, index, body)
+            entries.append(entry)
+            continue
+        entries.append(yaml_scalar(body))
+        index += 1
+    return entries, index

+def parse_command_front_matter(lines: list[str]) -> dict:
+    """Parse front-matter lines into one mapping (supported subset only)."""
+    data: dict = {}
+    index = 0
+    while index < len(lines):
+        line = lines[index]
+        if not line.strip() or line.lstrip().startswith("#"):
+            index += 1
+            continue
+        if line[0].isspace():
+            raise CommandDocError(f"unexpected indentation: {line!r}")
+        key, value = yaml_key_value(line, 0)
+        if key in data:
+            raise CommandDocError(f"duplicate key: {key}")
+        if value:
+            data[key] = yaml_scalar(value)
+            index += 1
+            continue
+        entries, index = yaml_list_block(lines, index + 1)
+        if not entries:
+            raise CommandDocError(f"key {key!r} needs a scalar value or a list")
+        data[key] = entries
+    return data
