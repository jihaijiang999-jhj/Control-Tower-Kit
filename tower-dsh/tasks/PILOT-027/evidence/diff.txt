+# Reproducing the tower test suite
+
+Clean-environment rerun checklist for `tower-dsh`.  `tools/run_tests.py` points
+`TMP`, `TEMP` and `TMPDIR` at a repository-local sandbox, so the suite also
+runs in a restricted DSH sandbox where the inherited `%TEMP%` directory carries
+ACLs that break `tempfile.mkdtemp`.
+
+## 1. Prerequisites
+
+- Python 3.12 (`python -V` → `Python 3.12.x`; last verified on 3.12.10).
+- Git (only used by the `.test-tmp` ignore test).
+- No network, no credentials, no `towerctl` state change: this is read-only
+  apart from the ignored `.test-tmp` sandbox.
+
+## 2. Exact command
+
+```sh
+cd tower-dsh
+python -B tools/run_tests.py
+```
+
+- `cwd` must be the tower root (the directory containing `tools/`).
+- Defaults: suite `tools/tests`, pattern `test*.py`, tmp-root
+  `<tower>/.test-tmp`.
+- The exit code is the child's exit code; `0` means green.
+
+Optional flags:
+
+```sh
+python -B tools/run_tests.py --suite tools/tests --pattern "test*.py"
+python -B tools/run_tests.py --suite <dir> --pattern test_smoke.py --tmp-root <dir>
+```
+
+Backward-compatible direct discovery (does **not** redirect temp; it uses the
+host `%TEMP%`/`$TMPDIR` and can fail under a restricted sandbox):
+
+```sh
+cd tower-dsh
+python -B -m unittest discover -s tools/tests -v
+```
+
+## 3. Where TMP lives
+
+The first stdout line names the sandbox:
+
+```text
+TOWER_TEST_TMP=<tower>/.test-tmp/run-YYYYmmdd-HHMMSS-<pid>
+```
+
+- The same directory is exported as `TMPDIR`, `TEMP` and `TMP` for the child
+  process; the rest of the environment is inherited unchanged.
+- The run directory is **never deleted** by the runner (a red run stays
+  debuggable).  It is gitignored via `tower-dsh/.test-tmp/`.
+- Each run directory carries a one-line `.git` gitfile that points at a
+  missing git dir.  The sandbox lives inside the verified repository, but the
+  fixtures assume temp directories are outside any repository; the gitfile
+  restores the old `not a git repository` result.  Fixtures that run `git init`
+  below the run directory are unaffected (the nearest `.git` wins).
+- On Windows the runner also exports `core.longpaths=true` through
+  `GIT_CONFIG_*` for its descendants: fixture repositories live deep under the
+  worktree and would otherwise hit `MAX_PATH` (`Filename too long`).
+- A relative `--tmp-root` resolves against the tower root, not the caller's
+  current directory.
+
+## 4. Expected result
+
+`Ran 246+ tests` followed by `OK`, exit code `0`.  On this branch (PILOT-027
+adds six runner tests) the count is 252:
+
+```text
+TOWER_TEST_TMP=.../.test-tmp/run-20261022-190837-4242
+...
+Ran 252 tests in 90.123s
+
+OK (skipped=1)
+```
+
+One skip is expected environment gating (it is reported, not silent).  Any
+failure or error is not expected.
+
+## 5. What to preserve on failure
+
+- Full stdout **and** stderr, including the `TOWER_TEST_TMP=` line.
+- The run directory it names — keep it until the failure is captured and
+  copied into evidence.
+- `python -V`, `git status --short`, and the failing test's name/traceback.
+
+## 6. Cleanup
+
+From the **repository root** (one level above `tower-dsh`):
+
+```sh
+python -c "import shutil; shutil.rmtree('tower-dsh/.test-tmp', ignore_errors=True)"
+```
+
+PowerShell: `Remove-Item -Recurse -Force tower-dsh\.test-tmp`
+POSIX: `rm -rf tower-dsh/.test-tmp`
+
+Cleanup is safe at any time: nothing under `.test-tmp` is tracked or needed
+after the evidence has been captured.

+#!/usr/bin/env python3
+"""Run the tower suite with ``TMP``/``TEMP``/``TMPDIR`` in a repo-local sandbox.
+
+Restricted DSH sandboxes break ``tempfile.mkdtemp`` because the inherited
+``%TEMP%`` directory carries ACLs the child cannot write through.  This runner
+creates one fresh directory per run under ``<tower>/.test-tmp`` and points all
+three temporary-directory variables at it, so tests and the code they exercise
+never touch the host temp directory.
+
+    python tools/run_tests.py
+    python tools/run_tests.py --suite tools/tests --pattern "test*.py"
+    python tools/run_tests.py --suite <dir> --pattern test_smoke.py --tmp-root <dir>
+
+The run directory is printed as ``TOWER_TEST_TMP=<dir>`` *before* the child
+starts, so a failing run can point straight at its own sandbox.  Run
+directories are never deleted (keeping them makes a red run debuggable); they
+are gitignored via ``tower-dsh/.test-tmp/``.
+
+The process exit code is the child's exit code.
+"""
+from __future__ import annotations
+
+import argparse
+import os
+import subprocess
+import sys
+from datetime import datetime
+from pathlib import Path
+
+# ``tools/run_tests.py`` -> the tower root is the directory holding ``tools/``.
+TOWER_ROOT = Path(__file__).resolve().parents[1]
+DEFAULT_SUITE = "tools/tests"
+DEFAULT_PATTERN = "test*.py"
+DEFAULT_TMP_ROOT = TOWER_ROOT / ".test-tmp"
+
+
+def build_parser() -> argparse.ArgumentParser:
+    parser = argparse.ArgumentParser(
+        description=("Run the tower unittest suite with a repo-local "
+                     "TMP/TEMP/TMPDIR sandbox"))
+    parser.add_argument(
+        "--suite", default=DEFAULT_SUITE,
+        help=f"test directory passed to unittest discover (default: {DEFAULT_SUITE})")
+    parser.add_argument(
+        "--pattern", default=DEFAULT_PATTERN,
+        help=f"unittest discovery pattern (default: {DEFAULT_PATTERN})")
+    parser.add_argument(
+        "--tmp-root", default=str(DEFAULT_TMP_ROOT),
+        help=("directory that holds each run sandbox "
+              f"(default: {DEFAULT_TMP_ROOT})"))
+    return parser
+
+
+def new_run_dir(tmp_root: Path) -> Path:
+    """Create and return ``<tmp-root>/run-YYYYmmdd-HHMMSS-<pid>``.
+
+    Existing directories are reused rather than failed on: a same-pid,
+    same-second rerun can only collide when the caller reuses a very stale
+    sandbox root, and reusing it is harmless.
+    """
+    stamp = datetime.now().strftime("%Y%m%d-%H%M%S")
+    run_dir = Path(tmp_root) / f"run-{stamp}-{os.getpid()}"
+    run_dir.mkdir(parents=True, exist_ok=True)
+    return run_dir
+
+
+def child_env(run_dir: Path) -> dict[str, str]:
+    """``os.environ`` with all three temp-dir variables pointed at ``run_dir``.
+
+    The sandbox sits inside the verified repository, while several fixtures
+    assume temporary directories are outside any git repository.  So the run
+    directory is also quarantined (see :func:`quarantine_run_dir`) and git's
+    ``core.longpaths`` is injected through ``GIT_CONFIG_*`` on Windows, because
+    throw-away fixture repositories under a deep worktree exceed ``MAX_PATH``.
+    Descendant git processes inherit both without touching global git config.
+    """
+    env = dict(os.environ)
+    for name in ("TMPDIR", "TEMP", "TMP"):
+        env[name] = str(run_dir)
+    if os.name == "nt":
+        _append_git_config(env, "core.longpaths", "true")
+    return env
+
+
+def _append_git_config(env: dict[str, str], key: str, value: str) -> None:
+    """Add one ``GIT_CONFIG_*`` entry without clobbering existing ones."""
+    try:
+        count = int(env.get("GIT_CONFIG_COUNT", "0") or 0)
+    except ValueError:
+        count = 0
+    env[f"GIT_CONFIG_KEY_{count}"] = key
+    env[f"GIT_CONFIG_VALUE_{count}"] = value
+    env["GIT_CONFIG_COUNT"] = str(count + 1)
+
+
+SANDBOX_GITFILE = "gitdir: ./.test-tmp-is-not-a-repository\n"
+
+
+def quarantine_run_dir(run_dir: Path) -> None:
+    """Stop git discovery from climbing out of the sandbox into the tower repo.
+
+    The sandbox lives inside the verified repository (``<tower>/.test-tmp``),
+    but the suite's fixtures assume a temporary directory is outside any git
+    repository: a few assert ``authority_dirty is None`` or expect git paths
+    to resolve to empty strings.  A gitfile in the run directory that points at
+    a missing git dir produces exactly the old ``not a git repository`` result
+    (exit 128) without changing any fixture.  Fixtures that run ``git init``
+    below the run directory are unaffected: the nearest ``.git`` wins.
+    """
+    gitfile = run_dir / ".git"
+    if not gitfile.exists():
+        gitfile.write_text(SANDBOX_GITFILE, encoding="utf-8")
+
+
+def main(argv: list[str] | None = None) -> int:
+    args = build_parser().parse_args(argv)
+    tmp_root = Path(args.tmp_root)
+    if not tmp_root.is_absolute():
+        tmp_root = TOWER_ROOT / tmp_root
+    run_dir = new_run_dir(tmp_root)
+    quarantine_run_dir(run_dir)
+    # Printed first and explicitly flushed: the child writes to the inherited
+    # stdout/stderr, so this line must not be buffered behind its output.
+    print(f"TOWER_TEST_TMP={run_dir}", flush=True)
+    completed = subprocess.run(
+        [sys.executable, "-B", "-m", "unittest", "discover",
+         "-s", args.suite, "-p", args.pattern, "-v"],
+        cwd=str(TOWER_ROOT),
+        env=child_env(run_dir),
+    )
+    return completed.returncode
+
+
+if __name__ == "__main__":
+    raise SystemExit(main())

+def run_process(argv, cwd, env=None, merge_streams=False, timeout=180):
+    """Run *argv* and always return decoded text output.
+
+    ``merge_streams`` sends stderr into stdout, which lets a test prove the
+    ``TOWER_TEST_TMP=`` line really is emitted *before* the child's output
+    rather than merely existing on a separate stream.
+    """
+    child_env = os.environ.copy()
+    child_env["PYTHONIOENCODING"] = "utf-8"
+    if env:

+    if merge_streams:
+        # stderr is folded into stdout so ordering between the two proves the
+        # banner was flushed before the child ran.
+        capture = {"stdout": subprocess.PIPE, "stderr": subprocess.STDOUT}
+    else:
+        capture = {"capture_output": True}
+    return subprocess.run(
+        argv,
+        cwd=str(cwd),
+        env=child_env,
+        text=True,
+        encoding="utf-8",
+        errors="replace",
+        timeout=timeout,
+        **capture,
